# CabinSense - dependencies (all models run locally; no paid APIs)

# IMPORTANT: install PyTorch + torchvision FIRST from the CUDA index matching your
# system, then install this file. For CPU-only use the default PyPI wheels.

#   pip install -r requirements.txt

# --- Deep learning / models ---
timm                          # backbones (MobileNetV3, MobileViT)
sixdrepnet                    # 6DRepNet robust head-pose (auto-downloads weights)
ultralytics                   # COCO YOLO object detection (phone, drinking); GPU via torch

# --- L2CS-Net gaze (not on PyPI; install from source) ---
# Run these AFTER the CUDA torch install, with --no-deps to protect the CUDA build:
#   pip install --no-deps git+https://github.com/Ahmednull/L2CS-Net.git
#   pip install --no-deps face-detection   # only to satisfy l2cs import (RetinaFace unused;
#                                           # our code shims it and uses YuNet for detection)
onnx                          # model export
onnxscript                    # required by torch.onnx.export (Torch 2.x exporter)
onnxruntime-gpu               # inference + INT8 quantization (onnxruntime.quantization)
huggingface_hub[hf_xet]       # pulling model/dataset artifacts

opencv-python                 # image/video IO, webcam, YuNet detector
pillow
